# Wallet/consensus separation gate probes (K1/W1; K3/W3 via check_symbols.py).
# Enabled only with -DDINERO_SEPARATION_GATES=ON. See
# docs/superpowers/specs/2026-10-29-wallet-node-separation-design.md section 8.
#
# Each probe links ONE root library whole-archive, plus exactly the link
# dependencies CMake propagates from that library (its INTERFACE_LINK_LIBRARIES)
# and nothing else. Whole-archive pulls in every object, so an undeclared
# dependency fails the link every time instead of depending on link order.
#
# Both targets are EXCLUDE_FROM_ALL so full builds stay green while the probes
# are RED; their ctests build the target first, then run it.

function(dinero_sep_forced_link target lib source)
  add_executable(${target} EXCLUDE_FROM_ALL ${source})
  target_compile_features(${target} PRIVATE cxx_std_20)
  # Plain link: brings ${lib} and its declared (propagated) deps, nothing more.
  target_link_libraries(${target} PRIVATE ${lib})
  if(MSVC)
    target_link_options(${target} PRIVATE "/WHOLEARCHIVE:$<TARGET_FILE_NAME:${lib}>")
  elseif(APPLE)
    # ld64: undefined symbols are already errors for executables.
    target_link_options(${target} PRIVATE "-Wl,-force_load,$<TARGET_FILE:${lib}>")
  else()
    target_link_options(${target} PRIVATE
      "-Wl,--whole-archive" "$<TARGET_FILE:${lib}>" "-Wl,--no-whole-archive"
      "-Wl,--no-undefined")
  endif()
endfunction()

dinero_sep_forced_link(sep_consensus_forced_link dinero_consensus consensus_forced_link.cpp)
dinero_sep_forced_link(sep_wallet_forced_link dinero_wallet wallet_forced_link.cpp)

set(_sep_driver ${CMAKE_CURRENT_SOURCE_DIR}/run_forced_link.cmake)
foreach(_pair "SepConsensusForcedLink;sep_consensus_forced_link"
              "SepWalletForcedLink;sep_wallet_forced_link")
  list(GET _pair 0 _test)
  list(GET _pair 1 _tgt)
  add_test(NAME ${_test} COMMAND ${CMAKE_COMMAND}
    -DSEP_BUILD_DIR=${CMAKE_BINARY_DIR}
    -DSEP_TARGET=${_tgt}
    -DSEP_CONFIG=$<CONFIG>
    -DSEP_EXE=$<TARGET_FILE:${_tgt}>
    -P ${_sep_driver})
  # One cmake --build per build dir at a time (ninja is not re-entrant);
  # later build-driving separation tests join the same lock.
  set_tests_properties(${_test} PROPERTIES LABELS "separation" TIMEOUT 1800
    RESOURCE_LOCK sep_build_dir)
endforeach()

# W6 standalone wallet-FFI consumer (spec section 8). Mirrors the Rust bridge
# link (mobile-tauri/src-tauri/build.rs:29-38): dinero_wallet_ffi whole-archive,
# then the static archives build.rs names, linked as plain FILES so CMake adds
# no transitive INTERFACE_LINK_LIBRARIES (no OpenSSL, no node libs). build.rs
# lists, beyond the four in the task brief: secp256k1, jsoncpp, sqlite3, c++,
# and (macOS) Security + Foundation, (Linux) dl + pthread. No
# target_link_libraries is used, so nothing is propagated.
if(TARGET dinero_wallet_ffi)
  add_executable(sep_wallet_ffi_standalone EXCLUDE_FROM_ALL wallet_ffi_standalone.cpp)
  target_compile_features(sep_wallet_ffi_standalone PRIVATE cxx_std_20)
  target_include_directories(sep_wallet_ffi_standalone PRIVATE
    ${CMAKE_SOURCE_DIR}/wallet-core/ffi)
  set(_sep_w6_deps dinero_wallet_ffi dinero_wallet dinero_crypto dinero_consensus
                   secp256k1 jsoncpp_static sqlite3)
  add_dependencies(sep_wallet_ffi_standalone ${_sep_w6_deps})
  set(_sep_w6_rest)
  foreach(_d dinero_wallet dinero_crypto dinero_consensus secp256k1 jsoncpp_static sqlite3)
    list(APPEND _sep_w6_rest "$<TARGET_FILE:${_d}>")
  endforeach()
  if(MSVC)
    target_link_options(sep_wallet_ffi_standalone PRIVATE
      "/WHOLEARCHIVE:$<TARGET_FILE_NAME:dinero_wallet_ffi>" "$<TARGET_FILE:dinero_wallet_ffi>"
      ${_sep_w6_rest})
  elseif(APPLE)
    target_link_options(sep_wallet_ffi_standalone PRIVATE
      "-Wl,-force_load,$<TARGET_FILE:dinero_wallet_ffi>" ${_sep_w6_rest}
      -lc++ "SHELL:-framework Security" "SHELL:-framework Foundation")
  else()
    target_link_options(sep_wallet_ffi_standalone PRIVATE
      "-Wl,--whole-archive" "$<TARGET_FILE:dinero_wallet_ffi>" "-Wl,--no-whole-archive"
      ${_sep_w6_rest} -lstdc++ -ldl -lpthread "-Wl,--no-undefined")
  endif()
  add_test(NAME SepWalletFfiStandalone COMMAND ${CMAKE_COMMAND}
    -DSEP_BUILD_DIR=${CMAKE_BINARY_DIR}
    -DSEP_TARGET=sep_wallet_ffi_standalone
    -DSEP_CONFIG=$<CONFIG>
    -DSEP_EXE=$<TARGET_FILE:sep_wallet_ffi_standalone>
    -P ${_sep_driver})
  # Label reflects the recorded baseline result: see README.md, W6.
  set_tests_properties(SepWalletFfiStandalone PROPERTIES
    LABELS "separation;recorded" TIMEOUT 1800 RESOURCE_LOCK sep_build_dir)
else()
  add_test(NAME SepWalletFfiStandalone COMMAND ${CMAKE_COMMAND} -E false)
  set_tests_properties(SepWalletFfiStandalone PROPERTIES LABELS "separation;recorded")
endif()

function(dinero_sep_wallet_fixture_target target source)
  # The wallet's Mempool seam (listUnspentUTXOs) resolves to the no-op test
  # stub, as in the Orchard wallet tests; no mempool is passed here.
  add_executable(${target} EXCLUDE_FROM_ALL ${source}
    ${CMAKE_SOURCE_DIR}/tests/stubs/mempool_test_stubs.cpp)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE
    ${CMAKE_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/third_party)
  target_compile_definitions(${target} PRIVATE
    SEP_WALLET_SCHEMA_PATH="${CMAKE_SOURCE_DIR}/resources/schema/wallet_schema.sql")
  # Same link line as test_descriptor_persistence (a wallet test that builds
  # today); the ON build adds the Orchard wallet-delivery owner.
  target_link_libraries(${target} PRIVATE dinero_wallet dinero_core dinero_crypto OpenSSL::SSL OpenSSL::Crypto)
  if(APPLE)
    target_link_libraries(${target} PRIVATE "-framework Security")
  endif()
  if(DINERO_BUILD_ORCHARD_BACKEND)
    target_compile_definitions(${target} PRIVATE SEP_FIXTURE_ORCHARD=1)
    target_link_libraries(${target} PRIVATE dinero_orchard_index_delivery)
  endif()
endfunction()
