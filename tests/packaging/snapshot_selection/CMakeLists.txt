cmake_minimum_required(VERSION 3.20)
project(DesktopSnapshotSelection LANGUAGES CXX)
enable_testing()
add_executable(snapshot_selection selection.cpp)
target_compile_features(snapshot_selection PRIVATE cxx_std_20)
set_target_properties(snapshot_selection PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(snapshot_selection PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../qt/src")
foreach(selection_case FreshPrimary MissingPrimary OldestOnly None Deduplicate DelimitedFallback WindowsAndUnicodePaths)
  add_test(NAME SnapshotSelection${selection_case} COMMAND snapshot_selection ${selection_case})
  set_tests_properties(SnapshotSelection${selection_case} PROPERTIES TIMEOUT 10)
endforeach()
